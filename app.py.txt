from datetime import datetime
import re
import bs4
import pandas as pd
import requests
import streamlit as st

st.set_page_config(
    page_title="オーストラリア中古車輸出 全自動適合判定",
    layout="wide",
    page_icon="🚗",
)

st.title("🚗 オーストラリア向け中古車輸出 全自動適合判定システム")
st.caption("SEVsデータベース ＋ ROVER Model Report ライブ自動照合ツール")


# 1. SEVデータの読み込み
@st.cache_data
def load_sev_data(file):
  df = pd.read_excel(file, skiprows=1)
  df.columns = [
      "SEV_No",
      "Make",
      "Model",
      "Category",
      "Model_Code",
      "Build_Date_From",
      "Build_Date_To",
      "Expiry",
  ]
  return df


# 2. ROVERからModel Report情報を自動取得する関数
def fetch_rover_mre_info(sev_no):
  """ROVERサイトへリクエストを送り、SEV Noに関連するModel Reportを検索する"""
  rover_url = (
      "https://www.rover.infrastructure.gov.au/PublishedApprovals/MREApprovals/"
  )
  headers = {
      "User-Agent": (
          "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML,"
          " like Gecko) Chrome/120.0.0.0 Safari/537.36"
      )
  }

  try:
    # ROVER検索用パラメーター
    params = {"RelatedApproval": sev_no}
    response = requests.get(
        rover_url, params=params, headers=headers, timeout=10
    )

    if response.status_code == 200:
      soup = bs4.BeautifulSoup(response.text, "html.parser")
      # テーブル要素の抽出処理
      table = soup.find("table")
      if table:
        rows = []
        for tr in table.find_all("tr")[1:]:  # ヘッダー以外
          cols = [td.text.strip() for td in tr.find_all("td")]
          if cols:
            rows.append(cols)
        if rows:
          return True, rows
      return False, "ROVER上に該当するModel Reportは見つかりませんでした。"
    else:
      return False, f"ROVER通信エラー (Status: {response.status_code})"
  except Exception as e:
    return False, f"自動取得エラー: {str(e)}"


# --- サイドバー画面構成 ---
st.sidebar.header("📁 1. データファイル")
uploaded_file = st.sidebar.file_uploader(
    "「AUS SEV早見表.xlsx」をアップロード", type=["xlsx", "xls"]
)

st.sidebar.header("🔍 2. 車両情報入力")
input_query = (
    st.sidebar.text_input("型式または車種名 (Model Code / Model)", value="AAHH40W")
    .strip()
    .upper()
)

col_y, col_m = st.sidebar.columns(2)
build_year = col_y.number_input(
    "製造年 (Build Year)", min_value=1980, max_value=2026, value=2024
)
build_month = col_m.number_input(
    "製造月 (Build Month)", min_value=1, max_value=12, value=1
)

raws_permission = st.sidebar.checkbox(
    "提携RAWs工場が Model Report の利用ライセンスを保有している", value=True
)

# 日付パース用ユーティリティ
def parse_my(date_str):
  if pd.isna(date_str) or str(date_str).strip() == "No end date":
    return None
  try:
    return datetime.strptime(str(date_str).strip(), "%m/%Y")
  except:
    return None


def parse_dmy(date_str):
  if pd.isna(date_str):
    return None
  try:
    return datetime.strptime(str(date_str).strip(), "%d/%m/%Y")
  except:
    return None


# --- 判定処理実行 ---
if st.sidebar.button("🚗 全自動適合判定を実行", type="primary"):
  if not uploaded_file:
    st.error("最初にサイドバーから「AUS SEV早見表.xlsx」をアップロードしてください。")
  elif not input_query:
    st.warning("型式または車種名を入力してください。")
  else:
    sev_df = load_sev_data(uploaded_file)
    target_date = datetime(build_year, build_month, 1)
    today = datetime.now()

    # あいまい部分一致検索
    query_clean = re.sub(r"\s+", "", input_query)

    def is_match(row):
      code = re.sub(r"\s+", "", str(row["Model_Code"]).upper())
      model = re.sub(r"\s+", "", str(row["Model"]).upper())
      return (
          (query_clean in code or code in query_clean) if code else False
      ) or ((query_clean in model or model in query_clean) if model else False)

    matched = sev_df[sev_df.apply(is_match, axis=1)]

    if matched.empty:
      st.error(
          f"❌ **輸出不可 (SEV未登録):** 入力された「{input_query}」に関連するSEV登録が見つかりませんでした。"
      )
    else:
      valid_sevs = []
      for idx, row in matched.iterrows():
        d_from = parse_my(row["Build_Date_From"])
        d_to = parse_my(row["Build_Date_To"])
        expiry = parse_dmy(row["Expiry"])

        in_range = True
        if d_from and target_date < d_from:
          in_range = False
        if d_to and target_date > d_to:
          in_range = False

        is_expired = expiry and expiry < today

        if in_range:
          valid_sevs.append((row, is_expired))

      if not valid_sevs:
        st.error(
            f"❌ **輸出不可 (製造年月対象外):** SEV登録 ({matched.iloc[0]['SEV_No']}) は存在しますが、指定の製造年月 ({build_year}/{build_month:02d}) は対象範囲外です。"
        )
      else:
        first_sev, is_expired = valid_sevs[0]
        sev_no = first_sev["SEV_No"]

        # --- ROVERサイトからの自動情報取得 ---
        st.subheader("🌐 3. ROVER Model Report リアルタイム照合結果")
        with st.spinner(f"ROVERサイトから {sev_no} の最新データを自動取得中..."):
          has_mre, mre_data = fetch_rover_mre_info(sev_no)

        if is_expired:
          st.warning(
              f"⚠️ **要確認 (SEV有効期限切れ):** SEV番号 `{sev_no}` の有効期限"
              f" ({first_sev['Expiry']}) が切れています。"
          )
        elif not has_mre:
          st.warning(
              f"⚠️ **判定保留 (Model Report未発見):** SEV適合 (`{sev_no}`)"
              " ですが、ROVER上でアクティブな Model Report"
              f" が確認できませんでした。({mre_data})"
          )
        elif not raws_permission:
          st.warning(
              f"⚠️ **判定保留 (RAWs利用権未確認):** SEV適合 (`{sev_no}`)"
              " および ROVER上に Model Report が確認されましたが、RAWs工場のライセンス状況を確認してください。"
          )
        else:
          st.success(
              f"✅ **輸出可能:** SEV適合 (`{sev_no}`)、製造年月適正、ROVER上の"
              " Model Report 情報照合完了！"
          )

        # 取得したMREテーブルデータの表示
        if has_mre and isinstance(mre_data, list):
          st.markdown("**【ROVER取得成功】関連する Model Report 一覧**")
          mre_df = pd.DataFrame(
              mre_data,
              columns=[
                  "Approval No",
                  "Approval Holder",
                  "Make / Model",
                  "Status",
                  "Expiry",
              ],
          )
          st.dataframe(mre_df, use_container_width=True)

        st.subheader("📋 該当SEVデータ")
        st.dataframe(matched, use_container_width=True)
